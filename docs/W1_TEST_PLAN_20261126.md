def _pixel_diff_rate(surf1: pygame.Surface, surf2: pygame.Surface) -> float:
    """计算两 Surface 像素差异率（RGB 距离 >10 视为不同）。"""
    diff_count = 0
    total_pixels = surf1.get_width() * surf1.get_height()
    for x in range(surf1.get_width()):
        for y in range(surf1.get_height()):
            color1 = surf1.get_at((x, y))
            color2 = surf2.get_at((x, y))
            rgb_dist = sum(abs(c1 - c2) for c1, c2 in zip(color1[:3], color2[:3]))
            if rgb_dist > 10:  # 阈值：RGB 距离 >10 视为不同
                diff_count += 1
    return diff_count / total_pixels
